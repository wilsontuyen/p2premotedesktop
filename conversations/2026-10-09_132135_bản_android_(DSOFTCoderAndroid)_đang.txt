373:     private fun receiveEncryptedJson(dis: DataInputStream): JSONObject {
374:         val length = dis.readInt()
375:         if (length <= 0 || length > 10 * 1024 * 1024) throw Exception("Invalid packet length: $length")
376:         
377:         val payload = ByteArray(length)
378:         dis.readFully(payload)
379:         
380:         if (payload.isNotEmpty() && payload[0] == '{'.code.toByte()) {
381:             return JSONObject(String(payload, Charsets.UTF_8))
382:         }
383: 
384:         if (length < 28) throw Exception("Packet too short for crypto")
385:         
386:         val decrypted = decrypt(payload)
387:         if (decrypted == null) throw Exception("Decryption failed")
388: 
389:         return JSONObject(String(decrypted, Charsets.UTF_8))
390:     }

392:     private fun decrypt(fullPayload: ByteArray): ByteArray? {
393:         if (fullPayload.size < 28) return null
394:         
395:         // Thử giải mã với khóa chính
396:         tinkCipher?.let { cipher ->
397:             try {
398:                 return cipher.decrypt(fullPayload, null)
399:             } catch (_: Exception) {}
400:         }
401: 
402:         // Thử giải mã với khóa dự phòng
403:         fallbackCipher?.let { cipher ->
404:             try {
405:                 return cipher.decrypt(fullPayload, null)
406:             } catch (_: Exception) {}
407:         }
408: 
409:         // Log lỗi nếu cả hai đều thất bại
410:         Log.e(TAG, "Decryption failed for payload of length ${fullPayload.size}")
411:         return null
412:     }

414:     private fun encrypt(plaintext: ByteArray): ByteArray? {
415:         val cipher = tinkCipher ?: return null
416:         return try {
417:             // Tink's encrypt(plaintext, associatedData) returns [Nonce(12)][Plaintext][Tag(16)]
418:             cipher.encrypt(plaintext, null)
419:         } catch (e: Exception) { 
420:             Log.e(TAG, "Encryption error: ${e.message}")
421:             null 
422:         }
423:     }

435:     private fun startFrameSender(s: Socket) {
436:         thread(name = "FrameSender") {
437:             try {
438:                 var lastSeenId = 0L
439:                 val cached = TcpStreamManager.instance?.peekLastGoodFrame()
440:                 if (cached != null) {
441:                     lastSeenId = cached.first
442:                     sendEncryptedData(s, cached.second)
443:                     Log.d(TAG, "FrameSender: reused cached frame id=${cached.first}")
444:                 } else {
445:                     Log.d(TAG, "FrameSender: no cache, waiting for first capture...")
446:                     val initialFrame = TcpStreamManager.instance?.waitForNextFrame(-1L, 800L)
447:                     if (initialFrame != null) {
448:                         lastSeenId = initialFrame.first
449:                         sendEncryptedData(s, initialFrame.second)
450:                     } else {
451:                         TcpStreamManager.instance?.forceRefresh()
452:                     }
453:                 }
454:                 
455:                 // Giới hạn số lần forceRefresh liên tiếp để tránh vòng lặp chết
456:                 // (đặc biệt trên Samsung khi recreate VirtualDisplay tốn tài nguyên)
457:                 var consecutiveRefreshCount = 0
458:                 val MAX_CONSECUTIVE_REFRESH = 3
459:                 
460:                 while (s.isConnected && !s.isClosed) {
461:                     val currentSocket = socket
462:                     if (currentSocket == null || currentSocket.isClosed) break
463: 
464:                     // Wait for next frame (timeout 2000ms to allow checking socket status)
465:                     val frameInfo = TcpStreamManager.instance?.waitForNextFrame(lastSeenId, 2000L)
466:                     if (frameInfo != null) {
467:                         lastSeenId = frameInfo.first
468:                         val jpegBytes = frameInfo.second
469:                         sendEncryptedData(s, jpegBytes)
470:                         consecutiveRefreshCount = 0  // Reset counter khi cA3 frame thAnh cA'ng
471:                     } else {
472:                         // HEARTBEAT RESEND: Nu Viewer b l> frame 'u tiAn (do mng hoc render li),
473:                         // chAAng ta gi li frame c sau mi 2 giAy cho d mAn hAnh khA'ng 'i
474:                         val cachedJpeg = TcpStreamManager.instance?.peekLastGoodFrame()?.second
475:                             ?: TcpStreamManager.instance?.getFrameJpeg()
476:                         if (cachedJpeg != null) {
477:                             Log.d(TAG, "No new frame. Resending cached frame (Heartbeat)...")
478:                             sendEncryptedData(s, cachedJpeg)
479:                         } else {
480:                             consecutiveRefreshCount++
481:                             if (consecutiveRefreshCount <= MAX_CONSECUTIVE_REFRESH) {
482:                                 // Fix for black screen on lock/unlock: force a refresh if the frame is stuck
483:                                 Log.d(TAG, "No frame, forceRefresh attempt $consecutiveRefreshCount/$MAX_CONSECUTIVE_REFRESH")
484:                                 TcpStreamManager.instance?.forceRefresh()
485:                             } else {
486:                                 if (consecutiveRefreshCount == MAX_CONSECUTIVE_REFRESH + 1) {
487:                                     Log.w(TAG, "Max forceRefresh reached. Waiting for frame without refresh...")
488:                                 }
489:                             }
490:                         }
491:                     }
492:                 }
493:             } catch (e: Exception) {
494:                 Log.e(TAG, "FrameSender error: ${e.message}")
495:             }
496:         }
497:     }

504:     private fun receiveLoop(s: Socket, dis: DataInputStream) {
505:         try {
506:             while (true) {
507:                 val length = dis.readInt()
508:                 if (length <= 0) break
509:                 if (length > 10 * 1024 * 1024) continue
510:                 
511:                 val payload = ByteArray(length)
512:                 dis.readFully(payload)
513:                 
514:                 if (payload.isNotEmpty() && payload[0] == '{'.code.toByte()) {
515:                     handlePeerMessage(String(payload, Charsets.UTF_8))
516:                     continue
517:                 }
518: 
519:                 val decrypted = decrypt(payload)
520:                 if (decrypted != null) {
521:                     val text = String(decrypted, Charsets.UTF_8)
522:                     if (text.startsWith("{")) {
523:                         if (text.length > 1000) Log.d(TAG, "Large message received: ${text.take(50)}...")
524:                         handlePeerMessage(text)
525:                     }
526:                 }
527:             }
528:         } catch (e: Exception) {
529:             Log.e(TAG, "receiveLoop error: ${e.message}")
530:         } finally { stop() }
531:     }

533:     fun sendEncryptedData(s: Socket, data: ByteArray) {
534:         try {
535:             if (s.isClosed || !s.isConnected) return
536:             
537:             val fullPayload = encrypt(data) ?: return
538:             
539:             val totalSize = fullPayload.size
540:             val buffer = ByteBuffer.allocate(4 + totalSize).order(ByteOrder.BIG_ENDIAN)
541:             buffer.putInt(totalSize)
542:             buffer.put(fullPayload)
543: 
544:             synchronized(this) {
545:                 val os = s.getOutputStream()
546:                 os.write(buffer.array())
547:                 os.flush()
548:             }
549:         } catch (e: Exception) {
550:             Log.e(TAG, "Send error: ${e.message}")
551:             if (e is java.net.SocketException || e.message?.contains("EPIPE") == true || e.message?.contains("reset") == true) {
552:                 stop()
553:             }
554:         }
555:     }
