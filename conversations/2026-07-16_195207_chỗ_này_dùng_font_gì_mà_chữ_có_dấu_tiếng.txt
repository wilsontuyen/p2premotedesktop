912:     def _do_format_partner_id(self):
913:         self._format_after_id = None
914:         
915:         # Save cursor position (now stable since key event is fully processed)
916:         try:
917:             cursor_pos = self.entry_p_id.index(tk.INSERT)
918:         except Exception:
919:             cursor_pos = None
920:         
921:         current_val = self.partner_id_var.get()
922:         raw_val = current_val.replace(" ", "")
923:         clean_val = "".join([c for c in raw_val if c.isdigit()])[:12]
924:         
925:         formatted = ""
926:         if len(clean_val) > 9:
927:             formatted = f"{clean_val[:3]} {clean_val[3:6]} {clean_val[6:9]} {clean_val[9:]}"
928:         elif len(clean_val) > 6:
929:             formatted = f"{clean_val[:3]} {clean_val[3:6]} {clean_val[6:]}"
930:         elif len(clean_val) > 3:
931:             formatted = f"{clean_val[:3]} {clean_val[3:]}"
932:         else:
933:             formatted = clean_val
934:         
935:         # Only update if the value actually changed
936:         if current_val == formatted:
937:             return
938:         
939:         # Count digits before cursor in the current (unformatted) string
940:         new_cursor = None
941:         if cursor_pos is not None:
942:             digits_before = sum(1 for c in current_val[:cursor_pos] if c.isdigit())
943:             # Find position in formatted string after the same number of digits
944:             count = 0
945:             new_cursor = len(formatted)
946:             for i, ch in enumerate(formatted):
947:                 if ch.isdigit():
948:                     count += 1
949:                     if count == digits_before:
950:                         new_cursor = i + 1
951:                         break
952:         
953:         self.partner_id_var.trace_remove("write", self.id_trace_id)
954:         self.partner_id_var.set(formatted)
955:         self.id_trace_id = self.partner_id_var.trace_add("write", self.format_partner_id)
956:         
957:         # Restore cursor position
958:         if new_cursor is not None:
959:             try:
960:                 self.entry_p_id.icursor(new_cursor)
961:             except Exception:
962:                 pass
